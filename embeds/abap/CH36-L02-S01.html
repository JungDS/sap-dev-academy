ZCL_BOOKING_AMDP (ADT에서 편집)
CLASS zcl_booking_amdp DEFINITION PUBLIC FINAL CREATE PUBLIC.
  PUBLIC SECTION.
    INTERFACES if_amdp_marker_hdb.
    " ts_stat/tt_stat TYPES 선언은 본문 코드와 동일(지면 생략)
    CLASS-METHODS get_stats
      IMPORTING VALUE(iv_client)  TYPE mandt
                VALUE(iv_concert) TYPE zconcert-concert_id
      EXPORTING VALUE(et_stats)   TYPE tt_stat
      AMDP OPTIONS READ-ONLY.
ENDCLASS.

CLASS zcl_booking_amdp IMPLEMENTATION.
  METHOD get_stats BY DATABASE PROCEDURE FOR HDB
                   LANGUAGE SQLSCRIPT
                   OPTIONS READ-ONLY
                   USING zbooking.
    et_stats =
      SELECT mandt, concert_id,
             COUNT(*) AS booking_cnt, SUM( seats ) AS total_seats
        FROM zbooking
       WHERE mandt = :iv_client
         AND concert_id = :iv_concert
       GROUP BY mandt, concert_id;
  ENDMETHOD.
ENDCLASS.
